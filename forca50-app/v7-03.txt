function lastSet(k){const a=[...S.sessions].sort((x,y)=>y.performed_at.localeCompare(x.performed_at)||String(y.created_at||'').localeCompare(String(x.created_at||'')));for(const s of a){const z=(s.sets||[]).filter(x=>x.exercise_key===k&&Number(x.reps)>0);if(z.length){const best=[...z].sort((x,y)=>Number(y.weight_kg)-Number(x.weight_kg)||Number(y.reps)-Number(x.reps))[0];return{session:s,set:best,sets:z}}}return null}
function progression(ex){const l=lastSet(ex.exercise_key);if(!l)return'Primeiro registro: escolha uma carga com RIR 1–2.';const top=Math.max(...l.sets.map(x=>Number(x.weight_kg)||0)),same=l.sets.filter(x=>(Number(x.weight_kg)||0)===top),hit=same.length>=Math.min(ex.sets_target,l.sets.length)&&same.every(x=>Number(x.reps)>=ex.reps_max);return`Último: ${top||'peso corporal'}${top?' kg':''} • ${hit&&top?`meta batida → tente ${(top+2.5).toFixed(1)} kg`:'repita a carga e busque mais reps'}`}
