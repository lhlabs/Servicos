function weekRows(n=8){const out=[],now=weekMonday();for(let i=n-1;i>=0;i--){const m=new Date(now);m.setDate(now.getDate()-i*7);const ds=weekDates(m).map(iso),count=trainedDaysInRange(ds);out.push({label:`${String(m.getDate()).padStart(2,'0')}/${String(m.getMonth()+1).padStart(2,'0')}`,count,start:iso(m)})}return out}
function attendancePct(n=8){const a=weekRows(n),t=weeklyTarget();return a.length?Math.round(a.reduce((s,x)=>s+Math.min(1,x.count/t),0)/a.length*100):0}

function renderFrequency(){const n=Number($('freqPeriod').value)||8,a=weekRows(n),t=weeklyTarget(),avg=a.reduce((s,x)=>s+x.count,0)/a.length,hit=a.filter(x=>x.count>=t).length,best=Math.max(0,...a.map(x=>x.count));let streak=0;for(let i=a.length-1;i>=0;i--){if(a[i].count>=t)streak++;else break}$('freqSummary').innerHTML=[['Média',avg.toFixed(1)+' dias'],['Meta atingida',`${hit}/${a.length}`],['Melhor semana',best+' dias'],['Sequência',streak+' sem.']].map(([l,v])=>`<div class="summarybox"><b>${v}</b><span>${l}</span></div>`).join('');$('frequencyBars').innerHTML=a.map(x=>`<div class="barrow"><span class="tiny">${x.label}</span><div class="bartrack"><div class="barfill ${x.count>=t?'hit':''}" style="width:${clamp(x.count/Math.max(7,t)*100,0,100)}%"></div></div><b>${x.count}</b></div>`).join('')}
$('freqPeriod').onchange=renderFrequency;
