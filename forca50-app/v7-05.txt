function exerciseNames(){const m=new Map();S.plan.exercises.forEach(x=>m.set(x.exercise_key,x.name));S.plan.substitutes.forEach(x=>m.set(x.substitute_key,x.name));S.sessions.forEach(s=>(s.sets||[]).forEach(x=>m.set(x.exercise_key,x.exercise_name)));return m}
function strengthLogs(k){const out=[];[...S.sessions].sort((a,b)=>a.performed_at.localeCompare(b.performed_at)).forEach(s=>{const a=(s.sets||[]).filter(x=>x.exercise_key===k&&Number(x.reps)>0);if(a.length)out.push({date:s.performed_at,e1rm:Math.max(...a.map(x=>e1(x.weight_kg,x.reps))),maxload:Math.max(...a.map(x=>Number(x.weight_kg)||0)),volume:a.reduce((z,x)=>z+(Number(x.weight_kg)||0)*(Number(x.reps)||0),0),reps:a.reduce((z,x)=>z+(Number(x.reps)||0),0)})});return out}
function periodFilter(a,p){if(p==='all')return a;const start=new Date();start.setDate(start.getDate()-Number(p));return a.filter(x=>dateAt(x.date)>=start)}
function drawChart(el,data,goal=null,unit='',label=''){
 if(!data.length){el.innerHTML='<div class="mut" style="padding:24px;text-align:center">Sem dados no período.</div>';return}
 const w=760,h=260,L=50,R=18,T=26,B=38,vals=data.map(x=>x.value).concat(goal==null?[]:[Number(goal)]),rawMin=Math.min(...vals),rawMax=Math.max(...vals),pad=Math.max((rawMax-rawMin)*.12,Math.abs(rawMax)*.02,1),min=rawMin-pad,max=rawMax+pad,range=max-min||1;
 const x=i=>data.length===1?(L+w-R)/2:L+i*(w-L-R)/(data.length-1),y=v=>T+(max-v)*(h-T-B)/range;
 let grid='';for(let i=0;i<5;i++){const v=max-i*range/4,yy=y(v);grid+=`<line x1="${L}" y1="${yy}" x2="${w-R}" y2="${yy}" stroke="#202a35"/><text x="${L-7}" y="${yy+4}" fill="#84909d" font-size="10" text-anchor="end">${formatValue(v,unit)}</text>`}
 const path=data.map((d,i)=>(i?'L':'M')+x(i).toFixed(1)+','+y(d.value).toFixed(1)).join(' '),goalLine=goal==null?'':`<line x1="${L}" y1="${y(goal)}" x2="${w-R}" y2="${y(goal)}" stroke="#ffd47c" stroke-width="2" stroke-dasharray="7 6"/><text x="${w-R}" y="${y(goal)-6}" fill="#ffd47c" font-size="10" text-anchor="end">meta ${formatValue(goal,unit)}</text>`;
 const idxs=[0,Math.floor((data.length-1)/2),data.length-1].filter((v,i,a)=>a.indexOf(v)===i),labels=idxs.map(i=>`<text x="${x(i)}" y="${h-11}" fill="#84909d" font-size="10" text-anchor="${i===0?'start':i===data.length-1?'end':'middle'}">${fmt(data[i].date).slice(0,5)}</text>`).join('');
 el.innerHTML=`<svg viewBox="0 0 ${w} ${h}" role="img" aria-label="${esc(label)}">${grid}${goalLine}<path d="${path}" fill="none" stroke="#87d4ff" stroke-width="3" stroke-linecap="round" stroke-linejoin="round"/>${data.map((d,i)=>`<circle cx="${x(i)}" cy="${y(d.value)}" r="${i===data.length-1?5:3.5}" fill="${i===data.length-1?'#d8ff63':'#87d4ff'}"><title>${fmt(d.date)} • ${formatValue(d.value,unit)}</title></circle>`).join('')}${labels}<text x="${L}" y="15" fill="#929ead" font-size="10">${esc(label)}</text></svg>`
}
function formatValue(v,u){if(u==='kgvol')return Math.round(v).toLocaleString('pt-BR');if(u==='reps')return Math.round(v).toLocaleString('pt-BR');return `${Number(v).toFixed(1)}${u?' '+u:''}`}
function summaryBoxes(el,items){el.innerHTML=items.map(([l,v])=>`<div class="summarybox"><b>${esc(v)}</b><span>${esc(l)}</span></div>`).join('')}
function renderStrength(){
 const key=$('progressExercise').value,metric=$('strengthMetric').value,p=$('strengthPeriod').value,unit=metric==='volume'?'kgvol':metric==='reps'?'reps':'kg',labels={e1rm:'e1RM',maxload:'Carga máxima',volume:'Volume',reps:'Repetições'},logs=periodFilter(strengthLogs(key),p).map(x=>({date:x.date,value:x[metric]}));if(!logs.length){summaryBoxes($('strengthSummary'),[['Atual','—'],['Melhor','—'],['Evolução','—'],['Registros','0']]);drawChart($('progressChart'),[],null,unit,labels[metric]);return}const first=logs[0].value,last=logs.at(-1).value,best=Math.max(...logs.map(x=>x.value)),chg=first?((last-first)/first*100):0;summaryBoxes($('strengthSummary'),[['Atual',formatValue(last,unit)],['Melhor',formatValue(best,unit)],['Evolução',`${chg>=0?'+':''}${chg.toFixed(1)}%`],['Registros',String(logs.length)]]);drawChart($('progressChart'),logs,null,unit,labels[metric])
}
function goalProgressFor(k){
 const g=S.goals?.[goalKeys[k]];if(g==null)return{goal:null};const a=[...S.checkpoints].filter(x=>x[k]!=null).sort((x,y)=>x.measured_at.localeCompare(y.measured_at));if(!a.length)return{goal:Number(g),current:null,base:null,pct:0,remaining:null,reached:false};const base=Number(a[0][k]),current=Number(a.at(-1)[k]),goal=Number(g),den=goal-base,raw=den===0?(current===goal?1:0):(current-base)/den,reached=den>=0?current>=goal:current<=goal;return{goal,current,base,pct:reached?100:clamp(raw*100,0,100),remaining:Math.abs(goal-current),reached}
}
function renderBody(){
 const k=$('bodyMetric').value,p=$('bodyPeriod').value,u=unitFor(k),goal=S.goals?.[goalKeys[k]],data=periodFilter([...S.checkpoints].filter(x=>x[k]!=null).sort((a,b)=>a.measured_at.localeCompare(b.measured_at)).map(x=>({date:x.measured_at,value:Number(x[k])})),p),gp=goalProgressFor(k);$('bodyGoalMini').textContent=goal!=null?`${num(goal)} ${u}`:'não definida';
 if(!data.length){summaryBoxes($('bodySummary'),[['Atual','—'],['Meta',goal!=null?`${num(goal)} ${u}`:'—'],['Falta','—'],['Progresso','—']]);drawChart($('bodyChart'),[],goal,u,bodyLabels[k]);return}const current=data.at(-1).value,first=data[0].value,delta=current-first;summaryBoxes($('bodySummary'),[['Atual',`${num(current)} ${u}`],['Meta',goal!=null?`${num(goal)} ${u}`:'—'],['Falta',gp.goal!=null&&gp.current!=null?(gp.reached?'atingida':`${num(gp.remaining)} ${u}`):'—'],['Progresso',gp.goal!=null?`${Math.round(gp.pct)}% • Δ ${delta>=0?'+':''}${num(delta)} ${u}`:`Δ ${delta>=0?'+':''}${num(delta)} ${u}`]]);drawChart($('bodyChart'),data,goal,u,bodyLabels[k])
}
function renderProgress(){
 const at=attendancePct(8),names=exerciseNames(),logged=[...names.keys()].filter(k=>strengthLogs(k).length);$('pSessions').textContent=S.sessions.length;$('pAttendance').textContent=`${at}%`;$('pVolume').textContent=Math.round(totalVolume()).toLocaleString('pt-BR');$('pExercises').textContent=logged.length;
 const sel=$('progressExercise'),old=sel.value;sel.innerHTML=logged.length?logged.map(k=>`<option value="${esc(k)}">${esc(names.get(k)||k)}</option>`).join(''):'<option value="">Sem registros</option>';if(logged.includes(old))sel.value=old;renderStrength();renderBody();
 const rows=[];for(const k of logged){const all=[];S.sessions.forEach(s=>(s.sets||[]).filter(x=>x.exercise_key===k&&Number(x.reps)>0).forEach(x=>all.push({...x,date:s.performed_at})));if(!all.length)continue;all.sort((a,b)=>b.date.localeCompare(a.date));const l=all[0],best=all.reduce((a,x)=>e1(x.weight_kg,x.reps)>e1(a.weight_kg,a.reps)?x:a,all[0]);rows.push({name:names.get(k),l,best})}$('recentLoads').innerHTML=rows.slice(0,30).map(x=>`<div class="loadrow"><div><b>${esc(x.name)}</b><div class="mut">último ${fmt(x.l.date)}</div></div><div style="text-align:right"><b>${num(x.l.weight_kg)} kg</b><div class="tiny">${x.l.reps} reps</div></div><div style="text-align:right"><b>${e1(x.best.weight_kg,x.best.reps).toFixed(1)}</b><div class="tiny">melhor e1RM</div></div></div>`).join('')||'<div class="mut" style="padding:18px 0">Registre cargas para acompanhar a evolução.</div>'
}
$('progressExercise').onchange=renderStrength;$('strengthMetric').onchange=renderStrength;$('strengthPeriod').onchange=renderStrength;$('bodyMetric').onchange=renderBody;$('bodyPeriod').onchange=renderBody;
