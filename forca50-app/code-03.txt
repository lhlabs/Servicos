  const ordered=[...S.sessions].sort((a,b)=>b.performed_at.localeCompare(a.performed_at)||String(b.created_at||'').localeCompare(String(a.created_at||'')));
  for(const s of ordered){const arr=(s.sets||[]).filter(x=>x.exercise_key===exKey&&Number(x.reps)>0);if(arr.length){const best=[...arr].sort((a,b)=>Number(b.weight_kg)-Number(a.weight_kg)||Number(b.reps)-Number(a.reps))[0];return{session:s,set:best,sets:arr}}}return null
}
function progression(ex){
  const l=lastSet(ex.exercise_key);if(!l)return'Primeiro registro: escolha uma carga com RIR 1–2.';
  const top=Math.max(...l.sets.map(x=>Number(x.weight_kg)||0));const same=l.sets.filter(x=>(Number(x.weight_kg)||0)===top);
  const hit=same.length>=Math.min(ex.sets_target,l.sets.length)&&same.every(x=>Number(x.reps)>=ex.reps_max);
  return `Último: ${top||'peso corporal'}${top?' kg':''} • ${hit&&top?`meta batida → tente ${(top+2.5).toFixed(1)} kg`:'repita a carga e busque mais reps'}`;
}
function attendanceStats(){
  const t=iso(), start=new Date();start.setDate(start.getDate()-55);start.setHours(0,0,0,0);let planned=0,done=0;
  for(let d=new Date(start);d<=new Date();d.setDate(d.getDate()+1)){const p=days().find(x=>x.day_of_week===d.getDay());if(!p||iso(d)>t)continue;planned++;if(sessionsOn(iso(d)).length)done++}
  return{planned,done,pct:planned?Math.round(done/planned*100):0}
}
